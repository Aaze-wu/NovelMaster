# -------------------------- 核心依赖（必需） --------------------------

# GUI 框架与内嵌浏览器
PyQt5>=5.15.0
PyQtWebEngine>=5.15.0

# EPUB 解析
ebooklib>=0.18
lxml>=4.9.0

# 图片处理（封面、主题图标）
Pillow>=10.0.0

# DOCX (Word) 解析
python-docx>=0.8.11

# 文本编码自动探测（TXT / HTML / FB2 等）
chardet>=5.0.0

# 界面暗色主题
qdarkstyle>=3.0.0

# PDF 解析（BSD-3-Clause）
pypdf>=6.0.0

# -------------------------- 可选依赖（非必需） --------------------------
# 以下依赖不安装也能正常运行，缺失时对应格式会自动降级或禁用。
#
# mobi>=0.4.1
#     MOBI / AZW / AZW3 / PRC 的增强解析库。
#     注意：该库采用 GPL-3.0-only 许可，与本项目的 MIT 许可不兼容，
#     因此不列为必需依赖。未安装时，程序使用内置的
#     PalmDB + PalmDOC 解析器读取上述格式，功能基本一致。
#
#     如确需安装（请自行评估许可影响）：
#     pip install mobi
#
# pywin32>=306
#     朗读功能的增强依赖。装上后朗读会走 sapi-com 后端（直接用 SAPI.SpVoice
#     驱动语音），能多枚举出系统 OneCore 语音库里的音色，例如中文的
#     Kangkang（男声）与 Yaoyao（女声）。未安装时自动回落到
#     PyQt5.QtTextToSpeech，朗读功能一样齐全，只是可选音色少一些。
#     官方发行版已随包附带，源码运行想体验更多音色时再装：
#     pip install pywin32
#
# sherpa-onnx>=1.13.8
#     朗读的「离线神经音色」依赖（Apache-2.0）。装上后可以用
#     Piper（超文 / 小雅 / 花颜）与 Kokoro（103 个音色，中英日等多语言）
#     这些神经网络音色，不联网、免费、音质比系统语音好很多。
#     注意：语音模型**不在**依赖里，第一次使用时在程序内
#     「朗读 → 音色管理…」里按需下载（Piper 小模型 13 MB，Kokoro 140 MB），
#     下完之后存在 %APPDATA%\NovelMaster\tts_models\ 下，可随时删除。
#     未安装时朗读自动回落系统语音，其余功能不受影响。
#     官方发行版已随包附带，源码运行想体验神经音色时再装：
#     pip install sherpa-onnx
#
# edge-tts>=7.2.8
#     朗读的「在线音色」依赖（MIT）。装上后可以挑选微软 Azure 的
#     300 多个在线音色（中文 6 个，如 Xiaoxiao / Yunxi / Xiaoyi）。
#     音质是三者里最好的，但需要联网，且音频是 MP3（用 QtMultimedia 播放）。
#     未安装时朗读自动回落系统语音；装了但断网时会明确提示合成失败。
#     pip install edge-tts
#
#     想省事就一次装齐：
#     pip install sherpa-onnx edge-tts
